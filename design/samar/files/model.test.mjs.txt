import {test} from 'node:test';import assert from 'node:assert/strict';import {normalize,schedule,clock,exportPlan} from './model.js';
test('invalid persisted plans cannot introduce unknown stops or impossible times',()=>{assert.deepEqual(normalize({ids:['registan','bad','registan','guri-amir'],start:'26:70'}),{ids:['registan','guri-amir'],start:'09:00'});assert.deepEqual(normalize('bad'),{ids:[],start:'09:00'});});
test('schedule respects user order, duration and transfer gaps',()=>{const rows=schedule({start:'09:30',ids:['guri-amir','registan']});assert.equal(rows[0].start,570);assert.equal(rows[0].end,630);assert.equal(rows[1].start,650);assert.equal(rows[1].end,725);});
test('late plans export the next day clearly and include source and estimate context',()=>{const p={start:'23:30',ids:['registan']};assert.equal(clock(schedule(p)[0].end),'00:45 (+1 kun)');const text=exportPlan(p);assert.match(text,/Registon/);assert.match(text,/https:\/\/uzbekistan.travel/);assert.match(text,/taklif/);});
